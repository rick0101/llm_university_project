```python
OUTPUT_PATH = "output/generated_figures/fig_poisoning_evolution_timeline.png"

import matplotlib.pyplot as plt
import numpy as np

# Data points provided in the specification
data = [
    {"label": "CGBA (Text)", "year": 2024, "category": "Text LLMs"},
    {"label": "CODEBREAKER (Code)", "year": 2024, "category": "Code LLMs"},
    {"label": "Persistent Backdoors", "year": 2024, "category": "Text LLMs"},
    {"label": "TrojVLM (Image-to-Text)", "year": 2024, "category": "Vision-Language Models"},
    {"label": "IAG (Visual Grounding)", "year": 2025, "category": "Vision-Language Models"},
    {"label": "VisualTrap (GUI Agents)", "year": 2025, "category": "AI Agents"},
    {"label": "BadVLA (VLA Systems)", "year": 2025, "category": "AI Agents"},
    {"label": "AutoBackdoor", "year": 2025, "category": "AI Agents"},
    {"label": "DecepChain (Reasoning)", "year": 2025, "category": "Reasoning Models"},
    {"label": "Breaking of Thought (BoT)", "year": 2025, "category": "Reasoning Models"},
    {"label": "SlowBA (Latency)", "year": 2026, "category": "AI Agents"},
    {"label": "BadTemplate (Infrastructure)", "year": 2026, "category": "Supply Chain"}
]

# Category color mapping for academic consistency
category_colors = {
    "Text LLMs": "#4C72B0",
    "Code LLMs": "#55A868",
    "Vision-Language Models": "#C44E52",
    "AI Agents": "#8172B3",
    "Reasoning Models": "#CCB974",
    "Supply Chain": "#64B5CD"
}

# Setup figure and style
plt.rcParams.update({
    "font.family": "serif",
    "font.serif": ["Times New Roman", "DejaVu Serif"],
    "font.size": 10,
    "axes.titlesize": 14,
    "axes.labelsize": 11
})

fig, ax = plt.subplots(figsize=(10, 6), constrained_layout=True)
ax.set_title("Evolution of Data and Model Poisoning Attacks (2024–2026)", pad=20, fontweight='bold')

# Timeline configuration
# We use levels to stagger the labels vertically to avoid overlap
# 2024 has 4 items, 2025 has 6 items, 2026 has 2 items
levels = np.array([-5, 5, -3, 3, -1, 1, -6, 6, -4, 4, -2, 2])

# Sort data by year to ensure logical progression
data.sort(key=lambda x: x['year'])

# Create the timeline
ax.axhline(0, color="black", linewidth=1.5, zorder=1)  # Main horizontal line

# Plot each point
for i, item in enumerate(data):
    year = item['year']
    # Add a slight horizontal jitter based on index within the same year to separate stems
    year_items = [d for d in data if d['year'] == year]
    item_idx_in_year = year_items.index(item)
    x_pos = year + (item_idx_in_year - len(year_items)/2 + 0.5) * 0.12
    
    level = levels[i % len(levels)]
    color = category_colors[item['category']]
    
    # Draw the stem
    ax.vlines(x_pos, 0, level, color="grey", linestyle="--", linewidth=0.8, alpha=0.6)
    
    # Draw the point on the timeline
    ax.scatter(x_pos, 0, color=color, s=80, zorder=3, edgecolors="white")
    
    # Add the label
    vert_alignment = "bottom" if level > 0 else "top"
    ax.text(x_pos, level + (0.2 if level > 0 else -0.2), item['label'],
            horizontalalignment="center", verticalalignment=vert_alignment,
            fontsize=9, fontweight='bold', color=color,
            bbox=dict(facecolor='white', edgecolor='none', alpha=0.7, pad=1))

# Formatting the axes
ax.set_xticks([2024, 2025, 2026])
ax.set_xticklabels(["2024", "2025", "2026 (Proj.)"], fontweight='bold')
ax.set_ylim(-8, 8)
ax.get_yaxis().set_visible(False)

# Remove spines
for spine in ["left", "top", "right"]:
    ax.spines[spine].set_visible(False)
ax.spines["bottom"].set_position("zero")

# Create a custom legend
from matplotlib.lines import Line2D
legend_elements = [Line2D([0], [0], marker='o', color='w', label=cat,
                          markerfacecolor=color, markersize=10)
                   for cat, color in category_colors.items()]
ax.legend(handles=legend_elements, loc='upper center', bbox_to_anchor=(0.5, -0.05),
          ncol=3, frameon=False, title="Target Modality / Domain")

# Add caption as a text box at the bottom
caption_text = "Figure 2: Chronological progression of state-of-the-art backdoor and poisoning attacks, illustrating the shift\nfrom text-based models to multimodal systems and autonomous agents."
plt.figtext(0.5, -0.15, caption_text, wrap=True, horizontalalignment='center', fontsize=10, style='italic')

# Save the figure
plt.savefig(OUTPUT_PATH, dpi=300, bbox_inches='tight')
plt.close()
```
